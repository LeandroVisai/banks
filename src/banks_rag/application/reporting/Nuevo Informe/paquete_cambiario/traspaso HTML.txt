def _last_year(frame: pd.DataFrame, fecha_col: str = 'Fecha') -> pd.DataFrame:
    """Filtra un DataFrame a los últimos 12 meses según fecha_col.

    Usado por todos los gráficos de soporte/resistencia para que los
    percentiles reflejen el régimen reciente y no toda la serie histórica.
    """
    max_date = frame[fecha_col].max()
    return frame[frame[fecha_col] >= max_date - pd.DateOffset(years=1)].copy()

def _add_pct_labels(fig: go.Figure, x_end, levels: list[tuple[float, str, str]]) -> None:
    """Tag de texto (P10, P25, …) junto al extremo de cada línea de percentil.

    Con la leyenda oculta (para no duplicar la info) esta es la única forma
    de identificar a simple vista qué línea es cada banda; el nivel exacto
    sigue viviendo en el hover.
    """
    for level, color, label in levels:
        fig.add_annotation(
            x=x_end, y=level, text=f'<b>{label}</b>',
            showarrow=False, xref='x', yref='y',
            xanchor='left', yanchor='middle', xshift=6,
            font=dict(size=9.5, color=color, family='Barlow, Arial, sans-serif'),
            bgcolor='rgba(255,255,255,0.80)', borderpad=1,
        )

# Monedas base 100:

# ───Funcion de calculo para las monedas en Base 100

def fig_base100_dynamic(df, fecha_col, series_cfg, title):
    """
    Gráfico de monedas base 100 con rebase dinámico por período.
    Cada botón (1M, 3M…) recalcula la base 100 desde el inicio de esa ventana.
    """
    max_date = df[fecha_col].dropna().max()

    # Fecha de inicio común para 'Todo': última primera-fecha-válida entre todas las series
    all_starts = [
        df[[fecha_col, col]].dropna(subset=[col])[fecha_col].min()
        for col, _, _ in series_cfg
        if len(df[[fecha_col, col]].dropna(subset=[col])) > 0
    ]
    min_date = max(all_starts) if all_starts else df[fecha_col].min()

    periods = [
        ('1M',  max_date - pd.DateOffset(months=1)),
        ('3M',  max_date - pd.DateOffset(months=3)),
        ('6M',  max_date - pd.DateOffset(months=6)),
        ('1A',  max_date - pd.DateOffset(years=1)),
        ('YTD', pd.Timestamp(max_date.year, 1, 1)),
        ('Todo', min_date),
    ]

    def _rebase(start):
        """Retorna (xs, ys) para todas las series rebased a 100 desde start."""
        xs, ys = [], []
        for col, _, _ in series_cfg:
            s = df[[fecha_col, col]].dropna(subset=[col]).sort_values(fecha_col)
            sf = s[s[fecha_col] >= start]
            if len(sf) == 0:
                xs.append([])
                ys.append([])
            else:
                base = sf[col].iloc[0]
                xs.append(sf[fecha_col].dt.strftime('%Y-%m-%d').tolist())
                ys.append((sf[col] / base * 100).tolist())
        return xs, ys

    # El gráfico muestra solo los últimos 2 meses (rebase a 100 desde el
    # inicio de esa ventana) en vez de toda la historia, para que se lea mejor.
    default_window_start = max(min_date, max_date - pd.DateOffset(months=2))
    init_xs, init_ys = _rebase(default_window_start)

    fig = go.Figure()
    for i, (_, label, color) in enumerate(series_cfg):
        fig.add_trace(go.Scatter(
            x=init_xs[i], y=init_ys[i],
            name=label,
            line=dict(color=color, width=1.5),
            hovertemplate=f'<b>{label}</b><br>%{{x|%Y-%m-%d}}: %{{y:.2f}}<extra></extra>',
        ))

    buttons = []
    for lbl, start in periods:
        xs, ys = _rebase(start)
        buttons.append(dict(
            label=lbl,
            method='update',
            args=[
                {'x': xs, 'y': ys},
                {'xaxis.range': [start.strftime('%Y-%m-%d'), max_date.strftime('%Y-%m-%d')]},
            ],
        ))

    fig.update_layout(
        **LAYOUT_BASE,
        title=dict(text=title, font=dict(size=15, color=NAVY)),
        xaxis=dict(**GRID, rangeslider=dict(visible=False)),
        yaxis=dict(**GRID, title='Índice Base 100'),
    )
    return fig

def _polish_hover(fig: go.Figure) -> go.Figure:
    """Uniforma el hover de los traces que no definieron hovertemplate propio.

    La mayoría de las figuras solo personalizan el hover de 1-2 series clave;
    el resto queda con el formato genérico de Plotly (fecha ISO larga, floats
    sin redondear). Acá se les da un formato consistente con el resto del
    dashboard (fecha corta, 2 decimales) sin pisar los hovertemplate ya
    definidos a mano. El candlestick usa su propio hover OHLC y se excluye.
    """
    for trace in fig.data:
        if trace.type == 'candlestick' or trace.hoverinfo == 'skip' or trace.hovertemplate:
            continue
        name = trace.name or ''
        trace.update(hovertemplate=f"<b>{name}</b><br>%{{x|%Y-%m-%d}}: %{{y:,.2f}}<extra></extra>")
    return fig

def _annotate_layout(groups: list, figs: dict) -> None:
    """Anota cada slot para que la grilla de 2 columnas nunca deje celdas vacías.

    Reglas (en orden):
    - pending: slot cuyo fig_id no existe en el registro de figuras -> tarjeta
      compacta a ancho completo, movida al final de la sección.
    - hero: si la sección lo declara, el primer gráfico real queda a ancho completo.
    - paridad: si la cantidad de gráficos restantes es impar, el último se
      estira a las 2 columnas para cerrar la fila.
    """
    for group in groups:
        for item in group["items"]:
            live, pending = [], []
            for slot in item["slots"]:
                slot["pending"] = slot.get("fig_id") not in figs
                (pending if slot["pending"] else live).append(slot)

            rest = live
            if item.get("hero") and live:
                live[0]["wide"] = True
                rest = live[1:]
            if len(rest) % 2 == 1:
                rest[-1]["wide"] = True

            item["slots"] = live + pending

/* ── PLOTLY LAZY RENDER ──
   Plotly mide su contenedor al construir, así que renderizamos cada sección
   recién antes de que entre en viewport (scrollspy), no toda de una vez.
*/
const __PLOTTED__ = new Set();

/* ── HELPERS DE PERÍODO / SLIDER / Y-SYNC ─────────────────────────── */

function _hasY2(el) {
  return !!(el.data && el.data.some(function(t) { return t.yaxis === 'y2'; }));
}

/* Calcula el rango de fechas real del chart a partir de sus datos */
function _getChartDateBounds(el) {
  if (!el.data) return null;
  var tMin = Infinity, tMax = -Infinity;
  for (var i = 0; i < el.data.length; i++) {
    var xs = el.data[i].x;
    if (!Array.isArray(xs)) continue;
    for (var j = 0; j < xs.length; j++) {
      if (!xs[j]) continue;
      var t = new Date(xs[j]).getTime();
      if (!isNaN(t)) { if (t < tMin) tMin = t; if (t > tMax) tMax = t; }
    }
  }
  return (tMin === Infinity) ? null : { min: tMin, max: tMax };
}

/* Inicializa el slider dual de fechas y lo vincula al chart */
function _initDateSlider(wrap, el) {
  var bounds = _getChartDateBounds(el);
  if (!bounds || bounds.max <= bounds.min) { wrap.style.display = 'none'; return; }

  var inp0  = wrap.querySelector('.date-slider-start');
  var inp1  = wrap.querySelector('.date-slider-end');
  var fill  = wrap.querySelector('.date-slider-fill');
  var lbl0  = wrap.querySelector('.date-slider-label-start');
  var lbl1  = wrap.querySelector('.date-slider-label-end');
  var STEPS = 1000;
  var span  = bounds.max - bounds.min;

  function valToTs(v)  { return bounds.min + (v / STEPS) * span; }
  function tsToVal(ts) { return Math.max(0, Math.min(STEPS, Math.round(((ts - bounds.min) / span) * STEPS))); }
  function fmtTs(ts) {
    return new Date(ts).toLocaleDateString('es-CL', { day: '2-digit', month: 'short', year: '2-digit' });
  }

  function updateUI() {
    var v0 = +inp0.value, v1 = +inp1.value;
    var lo = Math.min(v0, v1), hi = Math.max(v0, v1);
    fill.style.left  = (lo  / STEPS * 100) + '%';
    fill.style.width = ((hi - lo) / STEPS * 100) + '%';
    lbl0.textContent = fmtTs(valToTs(lo));
    lbl1.textContent = fmtTs(valToTs(hi));
  }

  function applyToChart() {
    var v0 = +inp0.value, v1 = +inp1.value;
    var lo = Math.min(v0, v1), hi = Math.max(v0, v1);
    var s0 = new Date(valToTs(lo)).toISOString().slice(0, 10);
    var s1 = new Date(valToTs(hi)).toISOString().slice(0, 10);
    var upd = { 'xaxis.range': [s0, s1], 'xaxis.autorange': false, 'yaxis.autorange': true };
    if (_hasY2(el)) upd['yaxis2.autorange'] = true;
    Plotly.relayout(el, upd);
    /* Desactiva los botones de período al usar el slider manualmente */
    var bar = document.querySelector('.period-bar[data-target="' + el.dataset.figId + '"]');
    if (bar) bar.querySelectorAll('.period-btn').forEach(function(b) { b.classList.remove('active'); });
  }

  inp0.addEventListener('input', function() { updateUI(); applyToChart(); });
  inp1.addEventListener('input', function() { updateUI(); applyToChart(); });

  inp0.value = 0; inp1.value = STEPS;
  updateUI();

  /* Sincroniza el slider cuando se activa un botón de período */
  el._syncSlider = function(startStr, endStr) {
    inp0.value = startStr ? tsToVal(new Date(startStr).getTime()) : 0;
    inp1.value = endStr   ? tsToVal(new Date(endStr).getTime())   : STEPS;
    updateUI();
  };
}

/* Aísla visualmente la serie bajo el cursor: la atenúa el resto de los traces
   (líneas, bandas, barras) mientras se hace hover y las restaura al salir.
   Con hovermode:'closest' el tooltip ya muestra solo la serie más cercana;
   esto agrega el refuerzo visual para que sea igual de claro en el gráfico. */
function _initHoverIsolate(el) {
  if (!el.data || el.data.length < 2) return; // 1 trace (ej. candlestick): nada que aislar
  var baseOpacities = el.data.map(function(t) { return (t.opacity == null) ? 1 : t.opacity; });
  var dimming = false;

  el.on('plotly_hover', function(ev) {
    if (!ev.points || !ev.points.length) return;
    var hoveredIdx = ev.points[0].curveNumber;
    dimming = true;
    Plotly.restyle(el, {
      opacity: el.data.map(function(_, i) { return i === hoveredIdx ? 1 : 0.10; }),
    });
  });

  el.on('plotly_unhover', function() {
    if (!dimming) return;
    dimming = false;
    Plotly.restyle(el, { opacity: baseOpacities });
  });
}

/* Llamar una vez por chart tras Plotly.newPlot */
function _postRender(el) {
  var sliderWrap = document.querySelector('.date-slider-wrap[data-target="' + el.dataset.figId + '"]');
  if (sliderWrap) _initDateSlider(sliderWrap, el);

  _initHoverIsolate(el);

  /* Sincroniza eje Y cuando cambia el rango X (zoom, etc.) */
  var _yTimer = null;
  el.on('plotly_relayout', function(ev) {
    var keys    = Object.keys(ev);
    var xChange = keys.some(function(k) { return k.startsWith('xaxis.range') || k === 'xaxis.autorange'; });
    var yChange = keys.some(function(k) { return k.startsWith('yaxis'); });
    if (xChange && !yChange) {
      clearTimeout(_yTimer);
      _yTimer = setTimeout(function() {
        var upd = { 'yaxis.autorange': true };
        if (_hasY2(el)) upd['yaxis2.autorange'] = true;
        Plotly.relayout(el, upd);
      }, 80);
    }
  });
}

/* Aplica un período al chart y sincroniza el slider */
function applyPeriodToEl(el, period) {
  if (!el || !el.data) return;
  if (el.dataset.rebase === 'true') { rebaseChart(el, period); return; }
  var startDate = _periodStartDate(period);
  var startStr  = startDate ? startDate.toISOString().slice(0, 10) : null;
  var endStr    = new Date().toISOString().slice(0, 10);
  var upd = startStr
    ? { 'xaxis.range': [startStr, endStr], 'xaxis.autorange': false, 'yaxis.autorange': true }
    : { 'xaxis.autorange': true, 'yaxis.autorange': true };
  if (_hasY2(el)) upd['yaxis2.autorange'] = true;
  Plotly.relayout(el, upd);
  if (el._syncSlider) el._syncSlider(startStr, startStr ? endStr : null);
}

function renderSection(secId) {
  if (!window.Plotly) return;
  const sec = document.getElementById('sec-' + secId);
  if (!sec) return;

  const newlyPlotted = [];
  sec.querySelectorAll('.plotly-target').forEach(el => {
    const id = el.dataset.figId;
    if (!id) return;
    const fig = window.__FIGS__[id];
    if (!fig) return;
    if (__PLOTTED__.has(id)) {
      Plotly.Plots.resize(el);
      return;
    }
    Plotly.newPlot(el, fig.data, fig.layout, {
      responsive: true,
      displayModeBar: 'hover',
      displaylogo: false,
      modeBarButtonsToRemove: ['lasso2d', 'select2d', 'autoScale2d']
    });
    __PLOTTED__.add(id);
    newlyPlotted.push(el);
  });

  // Post-render: slider + y-sync + período activo
  newlyPlotted.forEach(el => {
    _postRender(el);
    const figId = el.dataset.figId;
    const bar   = document.querySelector(`.period-bar[data-target="${figId}"]`);
    if (!bar) return;
    const active = bar.querySelector('.period-btn.active');
    if (!active || active.dataset.period === 'all') return;
    requestAnimationFrame(() => applyPeriodToEl(el, active.dataset.period));
  });
}

function canvasToBlob(canvas) {
  return new Promise((res, rej) =>
    canvas.toBlob(b => b ? res(b) : rej(new Error('blob fallido')), 'image/png')
  );
}
function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = filename;
  document.body.appendChild(a); a.click();
  setTimeout(() => { document.body.removeChild(a); URL.revokeObjectURL(url); }, 100);
}
function timestamp() {
  const d = new Date(), p = n => String(n).padStart(2, '0');
  return `${d.getFullYear()}${p(d.getMonth()+1)}${p(d.getDate())}_${p(d.getHours())}${p(d.getMinutes())}`;
}

/* ── CAPTURA POR GRÁFICO INDIVIDUAL → portapapeles ──
   Problema: .chart-card y .chart-area tienen overflow:hidden, lo que puede
   recortar el SVG de Plotly en la captura. Solución: quitar overflow
   temporalmente, capturar con dimensiones reales y restaurar.
*/
async function captureCard(btn) {
  const card  = btn.closest('.chart-card');
  const label = card.querySelector('.chart-title-left > span').textContent.trim();

  overlayMsg.textContent = 'Copiando gráfico…';
  overlay.classList.add('show');

  // 1) Guardar y neutralizar overflow que recorta
  const savedCard  = card.style.overflow;
  const area       = card.querySelector('.chart-area');
  const savedArea  = area ? area.style.overflow : null;
  card.style.overflow = 'visible';
  if (area) area.style.overflow = 'visible';

  try {
    // 2) Usar scrollHeight para capturar contenido completo aunque desborde
    const canvas = await html2canvas(card, {
      scale: 2,
      backgroundColor: '#ffffff',
      logging: false,
      useCORS: true,
      allowTaint: false,
      width:  card.offsetWidth,
      height: card.scrollHeight,
      windowWidth:  document.documentElement.scrollWidth,
      windowHeight: document.documentElement.scrollHeight,
    });

    // 3) Copiar al portapapeles
    const blob = await canvasToBlob(canvas);
    await navigator.clipboard.write([new ClipboardItem({ 'image/png': blob })]);
    showToast(`"${label}" copiado al portapapeles`);
    // Feedback en el propio botón: "Copiar" -> "Copiado" por 1.8s
    const lbl = btn.querySelector('.btn-capture-label');
    if (lbl && !btn._copiedTimer) {
      btn.classList.add('copied');
      lbl.textContent = 'Copiado';
      btn._copiedTimer = setTimeout(() => {
        btn.classList.remove('copied');
        lbl.textContent = 'Copiar';
        btn._copiedTimer = null;
      }, 1800);
    }
  } catch (err) {
    console.error(err);
    showToast('Error al copiar · ' + (err.message || err), true);
  } finally {
    // 4) Restaurar overflow siempre
    card.style.overflow = savedCard;
    if (area) area.style.overflow = savedArea;
    overlay.classList.remove('show');
  }
}

/* ── EXPORT ACTIONS ── */
async function exportAction(action) {
  // Target: el charts-grid de la sección activa (según scrollspy) para recorte preciso
  const activeNav = document.querySelector('.nav-item.active');
  const sec    = activeNav && document.getElementById('sec-' + activeNav.dataset.section);
  const target = sec && (sec.querySelector('.charts-grid') || sec);
  if (!target) return;

  overlayMsg.textContent = 'Copiando al portapapeles…';
  overlay.classList.add('show');

  // Neutralizar overflow:hidden para que html2canvas capture los SVG de Plotly completos
  const clippedEls    = Array.from(target.querySelectorAll('.chart-card, .chart-area'));
  const savedOverflows = clippedEls.map(el => el.style.overflow);
  clippedEls.forEach(el => (el.style.overflow = 'visible'));

  try {
    // Calcular altura hasta la última tarjeta con gráfico real (excluye placeholders vacíos)
    const liveCards = Array.from(target.querySelectorAll('.chart-card')).filter(c => c.querySelector('.plotly-target'));
    const gridTop   = target.getBoundingClientRect().top + window.scrollY;
    let captureH    = target.scrollHeight;
    if (liveCards.length > 0) {
      const lastRect = liveCards[liveCards.length - 1].getBoundingClientRect();
      captureH = Math.ceil(lastRect.bottom + window.scrollY - gridTop) + 12;
    }

    const canvas = await html2canvas(target, {
      scale: 2,
      backgroundColor: '#f5f3ee',
      logging: false,
      useCORS: true,
      allowTaint: false,
      width:  target.scrollWidth,
      height: captureH,
      windowWidth:  document.documentElement.scrollWidth,
      windowHeight: document.documentElement.scrollHeight,
    });
    const blob = await canvasToBlob(canvas);
    await navigator.clipboard.write([new ClipboardItem({ 'image/png': blob })]);
    showToast('Imagen copiada al portapapeles');
  } catch (err) {
    console.error(err);
    showToast('Error al exportar · ' + (err.message || err), true);
  } finally {
    clippedEls.forEach((el, i) => (el.style.overflow = savedOverflows[i]));
    overlay.classList.remove('show');
  }
}

/* ── PERÍODO GLOBAL ──
   Aplica el período seleccionado a TODOS los gráficos de la sección activa.
   - Charts con data-rebase="true": reindexan a base 100 desde el inicio del período.
   - Resto: solo ajustan xaxis.range.
*/

function _periodStartDate(period) {
  const now = new Date();
  if (period === '1m')  { const d = new Date(now); d.setMonth(d.getMonth() - 1); return d; }
  if (period === '3m')  { const d = new Date(now); d.setMonth(d.getMonth() - 3); return d; }
  if (period === '6m')  { const d = new Date(now); d.setMonth(d.getMonth() - 6); return d; }
  if (period === 'ytd') { return new Date(now.getFullYear(), 0, 1); }
  if (period === '1y')  { const d = new Date(now); d.setFullYear(d.getFullYear() - 1); return d; }
  return null; // 'all'
}

function rebaseChart(el, period) {
  if (!el || !el.data) return;
  const startDate = _periodStartDate(period);
  const startStr  = startDate ? startDate.toISOString().slice(0, 10) : null;
  const endStr    = new Date().toISOString().slice(0, 10);

  const newYs = el.data.map(trace => {
    const raw = trace.customdata;
    const xs  = trace.x;
    let baseIdx = 0;
    if (startStr) {
      for (let j = 0; j < xs.length; j++) {
        if ((xs[j] || '').slice(0, 10) >= startStr) { baseIdx = j; break; }
      }
    }
    const baseVal = raw[baseIdx];
    if (!baseVal) return raw.map(() => null);
    return raw.map(v => v != null ? +(v / baseVal * 100).toFixed(3) : null);
  });

  Plotly.restyle(el, { y: newYs });
  const rebaseUpd = startStr
    ? { 'xaxis.range': [startStr, endStr], 'xaxis.autorange': false, 'yaxis.autorange': true }
    : { 'xaxis.autorange': true, 'yaxis.autorange': true };
  Plotly.relayout(el, rebaseUpd);
}
